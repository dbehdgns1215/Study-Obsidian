```java
import java.io.*;
import java.util.*;

public class Solution {

    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringBuilder sb = new StringBuilder();
        StringTokenizer st;
        int T = Integer.parseInt(br.readLine());

        for (int tc = 1; tc <= T; tc++) {
            st = new StringTokenizer(br.readLine());
            int N = Integer.parseInt(st.nextToken());
            int W = Integer.parseInt(st.nextToken());

            int[] w = new int[N + 1];
            int[] v = new int[N + 1];
            
            for (int i = 1; i <= N; i++) {
                st = new StringTokenizer(br.readLine());
                w[i] = Integer.parseInt(st.nextToken());
                v[i] = Integer.parseInt(st.nextToken());
            }

            int[][] K = new int[N + 1][W + 1];

            for (int i = 1; i <= N; i++) {
                for (int weight = 0; weight <= W; weight++) {
                    if (w[i] > weight) {
                        K[i][weight] = K[i - 1][weight];
                    } else {
                        K[i][weight] = Math.max(v[i] + K[i - 1][weight - w[i]], K[i - 1][weight]);
                    }
                }
            }
            sb.append('#').append(tc).append(' ').append(K[N][W]).append('\n');
        }
        System.out.print(sb);
    }
}
```

# 위상정렬
```java
import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.LinkedList;
import java.util.Queue;
import java.util.StringTokenizer;

public class SWEA_1267 {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));

        for (int test_case = 1; test_case <= 10; test_case++) {
            StringTokenizer st = new StringTokenizer(br.readLine());

            int V = Integer.parseInt(st.nextToken()); // 정점 개수
            int E = Integer.parseInt(st.nextToken()); // 간선 개수

            Queue<Integer> q = new LinkedList<>();              // BFS를 돌기 위한 큐
            ArrayList<Integer>[] adj = new ArrayList[V + 1];    // 인접 리스트
            ArrayList<Integer> result = new ArrayList<>();      // 위상 정렬 결과

            for (int i = 0; i <= V; i++) {
                adj[i] = new ArrayList<>();
            }

            int[] deg = new int[V + 1]; // 각 정점의 진입 차수 저장

            st = new StringTokenizer(br.readLine());
            for (int i = 1; i <= E; i++) {
                int from = Integer.parseInt(st.nextToken());
                int to = Integer.parseInt(st.nextToken());
                adj[from].add(to); // 간선 저장
                deg[to]++;         // 진입 차수 증가
            }

            // 진입 차수가 0인 정점들을 큐에 추가
            for (int i = 1; i <= V; i++) {
                if (deg[i] == 0) q.add(i);
            }

            // 위상 정렬 수행
            while (!q.isEmpty()) {
                int cur = q.poll();
                result.add(cur);

                for (int next : adj[cur]) {
                    deg[next]--;
                    if (deg[next] == 0) q.add(next);
                }
            }

            // 출력
            StringBuilder sb = new StringBuilder();
            sb.append("#").append(test_case).append(" ");
            for (int i = 0; i < V; i++) {
                sb.append(result.get(i)).append(" ");
            }

            System.out.println(sb);
        }
    }
}
```

```java
package src;

import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;
import java.util.StringTokenizer;

public class SWEA_3124 {
    // Union-Find
    public static int[] parents;
    public static int V; // 정점 개수 (외부에서 초기화)

    // 초기화
    public static void make() {
        for (int i = 0; i <= V; i++) {
            parents[i] = -1; // Union By Rank (랭크 = 음수, 절댓값이 클수록 랭크 큼)
        }
    }

    // find: 루트 노드 찾기 (경로 압축)
    public static int find(int cur) {
        if (parents[cur] < 0) return cur; // 값 < 0 → 루트 노드
        return parents[cur] = find(parents[cur]); // 경로 압축
    }

    // union: 두 집합 합치기
    public static boolean union(int a, int b) {
        int aP = find(a);
        int bP = find(b);

        // 이미 같은 집합이면 합치지 않음
        if (aP == bP) return false;

        // 랭크 비교 (parents 값이 더 작을수록 랭크 큼)
        if (parents[aP] > parents[bP]) {
            // swap -> aP가 항상 랭크가 더 큰 쪽이 되도록
            int temp = aP;
            aP = bP;
            bP = temp;
        }

        // 랭크가 같으면 aP의 랭크 증가
        if (parents[aP] == parents[bP]) {
            parents[aP]--;
        }

        // bP를 aP의 자식으로 연결
        parents[bP] = aP;

        return true; // Union 연산 성공
    }
}
```

```java
public class SWEA_3124 {

    public static int[] parents;

    public static void make() {
        for (int i = 0; i <= V; i++) {
            parents[i] = -1; // Union By Rank (랭크 = 음수, 절댓값이 클수록 랭크 큼)
        }
    }

    public static int find(int cur) {

    }

    public static boolean union(int a, int b) {
        int aP = find(a);
        int bP = find(b);

        // 이미 같은 집합이면 합치지 않음
        if (aP == bP) return false;

        // 랭크 비교 (parents 값이 더 작을수록 랭크 큼)
        if (parents[aP] > parents[bP]) {
            // swap -> aP가 항상 랭크가 더 큰 쪽이 되도록
            int temp = aP;
            aP = bP;
            bP = temp;
        }

        // 랭크가 같으면 aP의 랭크 증가
        if (parents[aP] == parents[bP]) {
            parents[aP]--;
        }

        // bP를 aP의 자식으로 연결
        parents[bP] = aP;

        return true;
    }

    // --- 간선 클래스 (Kruskal에서 사용) ---
    public static class Edge implements Comparable<Edge> {
        int from, to, weight;

        public Edge(int from, int to, int weight) {
            this.from = from;
            this.to = to;
            this.weight = weight;
        }

        @Override
        public int compareTo(Edge o) {
            return Integer.compare(this.weight, o.weight);
        }
    }

    public static Edge[] edges;
    public static int V, E;

    // --- 메인 (Kruskal 알고리즘) ---
    public static void main(String[] args) throws NumberFormatException, IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringBuilder sb = new StringBuilder();
        StringTokenizer st;

        int T = Integer.parseInt(br.readLine());

        for (int test_case = 1; test_case <= T; test_case++) {
            st = new StringTokenizer(br.readLine());

            V = Integer.parseInt(st.nextToken());
            E = Integer.parseInt(st.nextToken());

            parents = new int[V + 1];
            edges = new Edge[E];

            for (int i = 0; i < E; i++) {
                st = new StringTokenizer(br.readLine());
                int from = Integer.parseInt(st.nextToken());
                int to = Integer.parseInt(st.nextToken());
                int weight = Integer.parseInt(st.nextToken());
                edges[i] = new Edge(from, to, weight);
            }

            Arrays.sort(edges);
            make();

            int cnt = 0;
            long sum = 0;

            for (Edge e : edges) {
                if (!union(e.from, e.to)) continue; // 사이클이면 skip
                sum += e.weight;
                if (++cnt == V - 1) break; // MST 완성
            }

            sb.append("#").append(test_case).append(" ").append(sum).append("\n");
        }

        System.out.print(sb);
    }
}
```

```java
class Node implements Comparable<Node> {
	int index;
	int cost;
	
	public Node(int index, int cost) {
		this.index = index;
		this.cost = cost;
	}

	@Override
	public int compareTo(Node o) {
		return Integer.compare(this.cost, o.cost);
	}
}

public class Main {
	static ArrayList<Node>[] graph;
	
    //노드의 크기, 출발지
	public static void Dijkstra(int n, int start) {
		boolean[] check = new boolean[n + 1];
		int[] dist = new int[n + 1];
		int INF = Integer.MAX_VALUE;
		
		Arrays.fill(dist, INF);
		dist[start] = 0;
		
		PriorityQueue<Node> pq = new PriorityQueue<>();
		pq.offer(new Node(start, 0));
		
		while(!pq.isEmpty()) {
			int nowVertex = pq.poll().index;
			
			if(check[nowVertex]) continue;
			check[nowVertex] = true;
			
			//index의 연결된 정점 비교
			for(Node next : graph[nowVertex]) {
				if(dist[next.index] > dist[nowVertex] + next.cost) {
					dist[next.index] = dist[nowVertex] + next.cost;
					
					pq.offer(new Node(next.index, dist[next.index]));
				}
			}
		}
        
        //최소거리 출력
		for(int i : dist) {
			if(i == INF) System.out.print(0 + " ");
			else System.out.print(i+" ");
		}
	}

	public static void main(String[] args) throws IOException {
    
    //그래프 입력 받기
		BufferedReader bf = new BufferedReader(new InputStreamReader(System.in));
		//정점의 개수, 간선의 개수
		int n = Integer.parseInt(bf.readLine());
		int m = Integer.parseInt(bf.readLine());
		
		graph = new ArrayList[n+1];
		for (int i = 0; i <= n; i++)  graph[i] = new ArrayList<>();
		
		StringTokenizer st;
		for(int i = 0 ; i < m; i++) {
			st = new StringTokenizer(bf.readLine());
			int v = Integer.parseInt(st.nextToken());
			int w = Integer.parseInt(st.nextToken());

		}

		int start = Integer.parseInt(bf.readLine());

		//다익스트라 알고리즘 수행
		Dijkstra(n, start);
		
	}
```

```java
import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.PriorityQueue;

public class SWEA_1249 {

    static int N;
    static final int INF = Integer.MAX_VALUE; // 최대 비용: 9 * 10000 = 90000 이하
    static int[][] map;
    static final int[] dx = {-1, 1, 0, 0};
    static final int[] dy = {0, 0, -1, 1};

    public static void main(String[] args) throws NumberFormatException, IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int T = Integer.parseInt(br.readLine());

        for (int tc = 1; tc <= T; tc++) {
            N = Integer.parseInt(br.readLine());
            map = new int[N][N];

            for (int i = 0; i < N; i++) {
                char[] row = br.readLine().toCharArray();
                for (int j = 0; j < N; j++) {
                    map[i][j] = row[j] - '0';
                }
            }

            System.out.println("#" + tc + " " + dijkstra(0, 0));
        }
    }

    private static int dijkstra(int startX, int startY) {
        int[][] minTime = new int[N][N];
		boolean[][] visited = new boolean[N][N];
		
		PriorityQueue<int[]> pq = new PriorityQueue<>((a, b) -> Integer.compare(a[2], b[2]));
		minTime[startY][startX] = 0;
		pq.offer(new int[]{startY, startX, 0});
		
		while (!pq.isEmpty()) {
		    int[] cur = pq.poll();
		    int y = cur[0];
		    int x = cur[1];
		    int cost = cur[2];
		
		    if (visited[y][x]) continue;
		    visited[y][x] = true;
		
		    if (y == N - 1 && x == N - 1) return cost;
		
		    for (int d = 0; d < 4; d++) {
		        int ny = y + dy[d];
		        int nx = x + dx[d];
		
		        if (ny >= 0 && ny < N && nx >= 0 && nx < N && !visited[ny][nx]) {
		            int newCost = cost + map[ny][nx];
		            if (newCost < minTime[ny][nx]) {
		                minTime[ny][nx] = newCost;
		                pq.offer(new int[]{ny, nx, newCost});
		            }
		        }
		    }
		}

        return -1;
    }
}
```

# 플로이드 워샬
```java
import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.StringTokenizer;

public class SWEA_1263 {

    private static final int INF = 1002; // 도달 불가를 의미 (문제 범위 내 충분히 큰 값)

    public static void main(String[] args) throws NumberFormatException, IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringBuilder sb = new StringBuilder();
        StringTokenizer st;

        int T = Integer.parseInt(br.readLine());

        for (int test_case = 1; test_case <= T; test_case++) {
            st = new StringTokenizer(br.readLine());

            int N = Integer.parseInt(st.nextToken());
            int[][] net = new int[N][N];

            // 인접 행렬 입력 -> 가중치 행렬로 변환
            for (int i = 0; i < N; i++) {
                for (int j = 0; j < N; j++) {
                    int a = Integer.parseInt(st.nextToken());
                    if (i == j) {
                        net[i][j] = 0;          // 자기 자신
                    } else if (a == 0) {
                        net[i][j] = INF;        // 간선 없음
                    } else {
                        net[i][j] = 1;          // 직접 연결
                    }
                }
            }

            // Floyd–Warshall
            for (int k = 0; k < N; k++) {
                for (int i = 0; i < N; i++) {
                    for (int j = 0; j < N; j++) {
                        if (net[i][j] > net[i][k] + net[k][j]) {
                            net[i][j] = net[i][k] + net[k][j];
                        }
                    }
                }
            }

            // 각 노드별 거리 합의 최소값
            int ans = Integer.MAX_VALUE;
            for (int i = 0; i < N; i++) {
                int sum = 0;
                for (int j = 0; j < N; j++) {
                    sum += net[i][j];
                }
                ans = Math.min(ans, sum);
            }

            sb.append("#").append(test_case).append(" ").append(ans).append("\n");
        }

        System.out.print(sb);
    }
}
```

```java
import java.util.Scanner;

public class DP1_KnapsackTest {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int N = sc.nextInt();  // 물건 개수
        int W = sc.nextInt();  // 배낭 최대 무게

        int[] weights = new int[N + 1];
        int[] profits = new int[N + 1];
        int[][] results = new int[N + 1][W + 1];

        // 물건 무게와 이익 입력
        for (int i = 1; i <= N; i++) {
            weights[i] = sc.nextInt();
            profits[i] = sc.nextInt();
        }

        // DP 계산
        for (int item = 1; item <= N; item++) {
            for (int weight = 1; weight <= W; weight++) {
                if (weights[item] <= weight) {
                    // i번 물건을 넣는 경우 vs 넣지 않는 경우 중 최대값
                    results[item][weight] = Math.max(
                        // ① 넣지 않는 경우
                        results[item - 1][weight],
                        // ② 넣는 경우
                        profits[item] + results[item - 1][weight - weights[item]]
                        
                    );
                } else {
                    // 배낭 무게보다 무거우면 넣을 수 없음
                    results[item][weight] = results[item - 1][weight]; // ③
                }
            }
        }

        // 최종 최대 이익
        System.out.println(results[N][W]);
        sc.close();
    }
}
```
