```python
# programmers_simple.py
import os
import time
import requests
import pandas as pd


BASE = "https://school.programmers.co.kr/api/v2/school/challenges/"

HEADERS = {
    "User-Agent": "Mozilla/5.0",
    "Accept": "application/json"
}

session = requests.Session()
session.headers.update(HEADERS)

# 기본 파라미터 (원하면 partIds[] 추가 가능)
per_page = 30
base_params = {"perPage": per_page, "order": "recent", "search": ""}

# 1) 첫 페이지 받아서 totalPages 확인
resp = session.get(BASE, params={**base_params, "page": 1}, timeout=10)
if resp.status_code != 200:
    print("첫 페이지 요청 실패:", resp.status_code)
    print(resp.text[:800])
    raise SystemExit(1)

j = resp.json()
total_pages = j.get("totalPages") or j.get("total_pages") or j.get("pageCount")
if not total_pages:
    # fallback: 계산
    total_entries = j.get("totalEntries") or (j.get("meta") or {}).get("total")
    if total_entries:
        total_pages = (int(total_entries) + per_page - 1) // per_page
    else:
        total_pages = 30  # 안전 fallback

print("총 페이지 수:", total_pages)

# 2) 모든 페이지 순회하여 result 합치기
all_rows = []
for page in range(1, int(total_pages) + 1):
    params = {**base_params, "page": page}
    print(f"Fetching page {page}/{total_pages} ...", end=" ")
    r = session.get(BASE, params=params, timeout=10)
    if r.status_code != 200:
        print("failed", r.status_code)
        print(r.text[:300])
        time.sleep(0.5)
        continue
    try:
        pj = r.json()
    except Exception as e:
        print("json decode error:", e)
        time.sleep(0.5)
        continue

    data = pj.get("result") or pj.get("data") or pj.get("challenges") or []
    if not isinstance(data, list):
        print("unexpected data on page", page)
        time.sleep(0.3)
        continue

    all_rows.extend(data)
    print("ok (got", len(data), "rows)")
    time.sleep(0.2)

# 3) DataFrame으로 저장 (원본 필드 그대로)
if all_rows:
    df = pd.DataFrame(all_rows)
    save_path = os.path.join("C:\\programmers", "programmers_results_simple.csv")
    df.to_csv(save_path, index=False, encoding="utf-8-sig")
    print("Saved:", save_path)
else:
    print("No data collected.")
```
