
# 위상정렬
```java
import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.LinkedList;
import java.util.Queue;
import java.util.StringTokenizer;

public class SWEA_1267 {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));

        for (int test_case = 1; test_case <= 10; test_case++) {
            StringTokenizer st = new StringTokenizer(br.readLine());

            int V = Integer.parseInt(st.nextToken()); // 정점 개수
            int E = Integer.parseInt(st.nextToken()); // 간선 개수

            Queue<Integer> q = new LinkedList<>();              // BFS를 돌기 위한 큐
            ArrayList<Integer>[] adj = new ArrayList[V + 1];    // 인접 리스트
            ArrayList<Integer> result = new ArrayList<>();      // 위상 정렬 결과

            for (int i = 0; i <= V; i++) {
                adj[i] = new ArrayList<>();
            }

            int[] deg = new int[V + 1]; // 각 정점의 진입 차수 저장

            st = new StringTokenizer(br.readLine());
            for (int i = 1; i <= E; i++) {
                int from = Integer.parseInt(st.nextToken());
                int to = Integer.parseInt(st.nextToken());
                adj[from].add(to); // 간선 저장
                deg[to]++;         // 진입 차수 증가
            }

            // 진입 차수가 0인 정점들을 큐에 추가
            for (int i = 1; i <= V; i++) {
                if (deg[i] == 0) q.add(i);
            }

            // 위상 정렬 수행 (Kahn’s Algorithm)
            while (!q.isEmpty()) {
                int cur = q.poll();
                result.add(cur);

                for (int next : adj[cur]) {
                    deg[next]--;
                    if (deg[next] == 0) q.add(next);
                }
            }

            // 출력
            StringBuilder sb = new StringBuilder();
            sb.append("#").append(test_case).append(" ");
            for (int i = 0; i < V; i++) {
                sb.append(result.get(i)).append(" ");
            }

            System.out.println(sb);
        }
    }
}
```
